
const express = require('express');
const axios = require('axios');
const fs = require('fs');
const app = express();
app.use(express.json());

// CONFIG - Se carga de config.json que exportas del bot limpio
let CONFIG = JSON.parse(fs.readFileSync('./config.json','utf8'));

// Variables de entorno - las pones en Render
const VERIFY_TOKEN = process.env.VERIFY_TOKEN || 'puntoytoma2026';
const WHATSAPP_TOKEN = process.env.WHATSAPP_TOKEN;
const PHONE_NUMBER_ID = process.env.PHONE_NUMBER_ID;
const MI_NUMERO_PERSONAL = process.env.MI_NUMERO_PERSONAL; // ej 5493546XXXXXX sin +

const sessions = {}; // memoria simple por cliente

function calcularPresupuesto(items, localidad) {
  let mano = 0, materiales = 0, detalle = [];
  const viaticoObj = CONFIG.viaticos.find(v => v.nombre === localidad) || {precio:0};
  for(let it of items){
    const serv = CONFIG.servicios.find(s => s.id === it.id);
    if(!serv) continue;
    mano += serv.precioMano * it.cantidad;
    materiales += (serv.precioMaterial || 0) * it.cantidad;
    detalle.push(`${it.cantidad}x ${serv.nombre} - $${serv.precioMano * it.cantidad}`);
  }
  const subtotal = mano + viaticoObj.precio;
  return {detalle, mano, viatico: viaticoObj, materiales, totalMano: subtotal, totalConMateriales: subtotal + materiales};
}

async function enviarWhatsApp(to, text, buttons=null){
  const url = `https://graph.facebook.com/v20.0/${PHONE_NUMBER_ID}/messages`;
  let data = {
    messaging_product: "whatsapp",
    to,
    type: "text",
    text: {body: text}
  };
  if(buttons){
    data = {
      messaging_product: "whatsapp",
      to,
      type: "interactive",
      interactive: {
        type: "button",
        body: {text},
        action: {buttons: buttons.map(b=>({type:"reply", reply:{id:b.id, title:b.title}}))}
      }
    }
  }
  await axios.post(url, data, {headers: {Authorization: `Bearer ${WHATSAPP_TOKEN}`}});
}

// WEBHOOK VERIFICACION
app.get('/webhook', (req,res)=>{
  if(req.query['hub.verify_token'] === VERIFY_TOKEN){
    return res.send(req.query['hub.challenge']);
  }
  res.sendStatus(403);
});

// WEBHOOK MENSAJES
app.post('/webhook', async (req,res)=>{
  res.sendStatus(200);
  try{
    const msg = req.body.entry?.[0]?.changes?.[0]?.value?.messages?.[0];
    if(!msg) return;
    const from = msg.from;
    const text = msg.text?.body?.toLowerCase() || msg.button?.payload || msg.interactive?.button_reply?.id || '';

    if(!sessions[from]) sessions[from] = {step:0, items:[], localidad: null};

    const s = sessions[from];

    // Flujo simple
    if(s.step === 0){
      let listaServ = CONFIG.servicios.map(sv => `- ${sv.nombre}: $${sv.precioMano}`).join('\n');
      if(CONFIG.servicios.length===0) listaServ = "(Aún no cargaste servicios en el panel, hablá con Cristian)";
      await enviarWhatsApp(from, `Hola! ⚡ Soy el asistente de *Punto Y Toma*\n\n¿En qué te ayudo?\n${listaServ}\n\nEscribí el nombre del servicio que querés, ej: "2 tomas" o "5m guirnalda"`);
      s.step = 1;
      return;
    }

    if(text.includes('cristian') || text.includes('humano')){
      await enviarWhatsApp(from, "Perfecto, te derivo con Cristian C. en un momento te escribe.");
      // Avisa a Cristian
      if(MI_NUMERO_PERSONAL){
        await enviarWhatsApp(MI_NUMERO_PERSONAL, `🔔 *Nuevo cliente pide hablar*\nDe: ${from}\nQuiso: hablar con humano`);
      }
      delete sessions[from];
      return;
    }

    // Intenta parsear pedido simple: ej "2 tomas" "5m guirnalda"
    // Por ahora acumulamos todo como texto y calculamos con lo que encuentre
    // Para prueba, si escribe "presupuesto" cerramos
    if(text.includes('presupuesto') || text.includes('cuanto')){
      // calcula con items
      if(s.items.length===0 && CONFIG.servicios.length>0){
        // demo: toma primer servicio x1
        s.items = [{id: CONFIG.servicios[0].id, cantidad:1}];
      }
      const calc = calcularPresupuesto(s.items, s.localidad || "Villa General Belgrano");
      let detalleTxt = calc.detalle.join('\n');
      let mensaje = `🧾 *Presupuesto Punto Y Toma*\n\n${detalleTxt}\nViático ${calc.viatico.nombre||'VGB'}: $${calc.viatico.precio}\n\n*Mano de obra: $${calc.totalMano}*\n${calc.materiales>0?`Materiales estimados: $${calc.materiales}\nTotal con materiales: $${calc.totalConMateriales}\n`:''}\nValidez 7 días. Precio accesible ajustable en visita.\n\nAhora te comunico con Cristian para coordinar.`;
      await enviarWhatsApp(from, mensaje);
      
      if(MI_NUMERO_PERSONAL){
        await enviarWhatsApp(MI_NUMERO_PERSONAL, `⚡ *NUEVO PRESUPUESTO BOT*\nCliente: ${from}\n${detalleTxt}\nZona: ${calc.viatico.nombre} $${calc.viatico.precio}\nTotal MO: $${calc.totalMano} | Con mat: $${calc.totalConMateriales}\n\nEl cliente ya recibió el presupuesto. Escribile!`);
      }
      delete sessions[from];
      return;
    }

    // Guardar localidad si la detecta
    const loc = CONFIG.viaticos.find(v => text.includes(v.nombre.toLowerCase()));
    if(loc){ s.localidad = loc.nombre; await enviarWhatsApp(from, `Zona ${loc.nombre} registrada (viático $${loc.precio}). Ahora decime qué servicio necesitas.`); return; }

    // Agregar item si menciona servicio
    for(let serv of CONFIG.servicios){
      if(text.includes(serv.nombre.toLowerCase().split(' ')[0])){
        let cant = parseInt(text) || 1;
        s.items.push({id: serv.id, cantidad: cant});
        await enviarWhatsApp(from, `Agregado: ${cant}x ${serv.nombre}. ¿Algo más? Escribí otro servicio o escribí PRESUPUESTO para cerrar.`);
        return;
      }
    }

    await enviarWhatsApp(from, `No entendí bien. Escribí por ejemplo "2 tomas y 3m guirnalda" o tu zona (Santa Rosa, etc). Si querés hablar conmigo escribí CRISTIAN.`);

  }catch(e){ console.error(e.response?.data || e.message); }
});

app.get('/', (req,res)=> res.send('Punto Y Toma Bot activo ⚡'));

app.listen(process.env.PORT||3000, ()=> console.log('Bot corriendo'));
